  // State Management
  const [activeScenario, setActiveScenario] = useState('nominal');
  const [vitals, setVitals] = useState(SCENARIOS.nominal.vitals);
  const [selectedSystemId, setSelectedSystemId] = useState('neuro');
  const [selectedChartMetric, setSelectedChartMetric] = useState('icp_bp');
  const [tasks, setTasks] = useState(INITIAL_TASKS);
  const [streakSols, setStreakSols] = useState(44);
  const [isLogModalOpen, setIsLogModalOpen] = useState(false);
  const [isExamModalOpen, setIsExamModalOpen] = useState(false);
  const [examStepProgress, setExamStepProgress] = useState(0);
  const [lastSyncTime, setLastSyncTime] = useState('14:22:09 UTC');
  const [notificationBanner, setNotificationBanner] = useState(null);

  // Switch Scenario Handler
  const handleSelectScenario = (scenarioKey) => {
    const sc = SCENARIOS[scenarioKey];
    setActiveScenario(scenarioKey);
    setVitals({ ...sc.vitals });

    // Automatically select the most affected system to highlight immediate value
    if (scenarioKey === 'radiation') setSelectedSystemId('radiation');
    else if (scenarioKey === 'sans') setSelectedSystemId('neuro');
    else if (scenarioKey === 'isolation') setSelectedSystemId('cardio');
    else setSelectedSystemId('neuro');

    // Reset some tasks when emergency scenarios trigger so user can test fixing them
    if (scenarioKey !== 'nominal') {
      setTasks(prev => prev.map((t, i) => ({ ...t, completed: i === 0 })));
    }
  };

  // Toggle Countermeasure Task & Apply Immediate Physiological Improvement
  const handleToggleTask = (taskId) => {
    setTasks(prev =>
      prev.map(task => {
        if (task.id !== taskId) return task;
        const nextState = !task.completed;

        // Apply simulated bio-feedback adjustments when astronaut acts on countermeasures
        const delta = nextState ? 1 : -1;
        setVitals(v => {
          const updated = { ...v };
          if (taskId === 'lbnp') {
            updated.icp = Number((v.icp - 1.8 * delta).toFixed(1));
            updated.opticNerveDiameter = Number((v.opticNerveDiameter - 0.3 * delta).toFixed(1));
          } else if (taskId === 'ared') {
            updated.urinaryCalcium = Math.round(v.urinaryCalcium - 35 * delta);
            updated.aredCompliance = Math.min(100, Math.max(50, v.aredCompliance + 12 * delta));
          } else if (taskId === 't2') {
            updated.glucose = Math.round(v.glucose - 9 * delta);
            updated.hrv = Math.round(v.hrv + 6 * delta);
          } else if (taskId === 'sla_light') {
            updated.sleepHours = Number(Math.min(9, v.sleepHours + 0.9 * delta).toFixed(1));
            updated.cortisol = Number((v.cortisol - 2.8 * delta).toFixed(1));
          } else if (taskId === 'nutri') {
            updated.wbc = Number((v.wbc + 0.5 * delta).toFixed(1));
            updated.vitaminD = Math.round(v.vitaminD + 6 * delta);
          } else if (taskId === 'vr_mind') {
            updated.moodScore = Number(Math.min(10, v.moodScore + 1.3 * delta).toFixed(1));
            updated.systolic = Math.round(v.systolic - 7 * delta);
            updated.heartRate = Math.round(v.heartRate - 4 * delta);
          }
          return updated;
        });

        if (nextState) {
          setNotificationBanner(`Countermeasure Executed: ${task.title} (${task.impactText})`);
          setTimeout(() => setNotificationBanner(null), 4000);
        }

        return { ...task, completed: nextState };
      })
    );
  };

  // Evaluate System Status (Nominal / Caution / Critical)
  const systemStatuses = useMemo(() => {
    const getStatus = (sysId) => {
      if (sysId === 'neuro') {
        if (vitals.icp >= 18 || vitals.opticNerveDiameter >= 6.0 || vitals.sleepHours < 5.2 || vitals.moodScore < 5.0) return 'CRITICAL';
        if (vitals.icp >= 15 || vitals.opticNerveDiameter >= 5.5 || vitals.sleepHours < 6.5 || vitals.moodScore < 6.8) return 'CAUTION';
        return 'NOMINAL';
      }
      if (sysId === 'cardio') {
        if (vitals.systolic >= 140 || vitals.diastolic >= 90 || vitals.hrv < 42) return 'CRITICAL';
        if (vitals.systolic >= 128 || vitals.heartRate >= 74 || vitals.hrv < 55) return 'CAUTION';
        return 'NOMINAL';
      }
      if (sysId === 'immune') {
        if (vitals.wbc < 3.8 || vitals.cortisol > 22 || vitals.bodyTemp >= 37.8) return 'CRITICAL';
        if (vitals.wbc < 4.5 || vitals.cortisol > 16 || vitals.bodyTemp >= 37.3) return 'CAUTION';
        return 'NOMINAL';
      }
      if (sysId === 'metabolic') {
        if (vitals.glucose >= 120 || vitals.hydration < 80 || vitals.vitaminD < 25) return 'CRITICAL';
        if (vitals.glucose >= 102 || vitals.hydration < 88 || vitals.vitaminD < 36) return 'CAUTION';
        return 'NOMINAL';
      }
      if (sysId === 'musculo') {
        if (vitals.bmdTScore <= -1.2 || vitals.urinaryCalcium >= 280) return 'CRITICAL';
        if (vitals.bmdTScore <= -0.6 || vitals.urinaryCalcium >= 220) return 'CAUTION';
        return 'NOMINAL';
      }
      if (sysId === 'radiation') {
        if (vitals.dailyRadRate >= 2.0 || vitals.cumRadiation >= 250) return 'CRITICAL';
        if (vitals.dailyRadRate >= 0.6 || vitals.cumRadiation >= 150) return 'CAUTION';
        return 'NOMINAL';
      }
      return 'NOMINAL';
    };

    return {
      neuro: getStatus('neuro'),
      cardio: getStatus('cardio'),
      immune: getStatus('immune'),
      metabolic: getStatus('metabolic'),
      musculo: getStatus('musculo'),
      radiation: getStatus('radiation'),
    };
  }, [vitals]);

  // Calculate Crew Bio-Readiness Score (0 - 100)
  const bioReadinessScore = useMemo(() => {
    let score = 96;
    Object.values(systemStatuses).forEach(st => {
      if (st === 'CRITICAL') score -= 14;
      if (st === 'CAUTION') score -= 6;
    });
    const completedCount = tasks.filter(t => t.completed).length;
    score += completedCount * 1.5;
    return Math.max(32, Math.min(99, Math.round(score)));
  }, [systemStatuses, tasks]);

  // Total XP
  const totalXP = useMemo(() => {
    const baseXP = 4250;
    const taskXP = tasks.filter(t => t.completed).reduce((acc, t) => acc + t.xp, 0);
    return baseXP + taskXP;
  }, [tasks]);

  // Dynamic Autonomous Countermeasure Prescriptions based on current vitals
  const prioritizedActions = useMemo(() => {
    const actions = [];

    if (vitals.dailyRadRate >= 1.5 || vitals.wbc < 4.0) {
      actions.push({
        id: 'act-rad',

      });
    }

    if (vitals.icp >= 16.0 || vitals.opticNerveDiameter >= 5.7) {
      actions.push({
        id: 'act-sans',
        priority: 'HIGH PRIORITY',
        system: 'Neurological / SANS',
        title: 'Initiate 45-Min Lower Body Negative Pressure (LBNP) @ -40 mmHg',
        detail: `Elevated Intracranial Pressure (${vitals.icp} mmHg) & Optic Nerve Sheath (${vitals.opticNerveDiameter} mm) indicate cephalad venous engorgement.`,
        linkedTaskId: 'lbnp',
        badgeColor: 'bg-amber-500/20 text-amber-300 border-amber-500/50'
      });
    }

    if (vitals.urinaryCalcium >= 240 || vitals.bmdTScore <= -0.8) {
      actions.push({
        id: 'act-bone',
        priority: 'HIGH PRIORITY',
        system: 'Musculoskeletal',
        title: 'Increase ARED Resistive Load +10% & Increase Hydration to 3.2L/day',
        detail: `Urinary calcium (${vitals.urinaryCalcium} mg/d) & BMD T-Score (${vitals.bmdTScore}) indicate accelerated osteoclastic resorption and renal stone risk.`,
        linkedTaskId: 'ared',
        badgeColor: 'bg-amber-500/20 text-amber-300 border-amber-500/50'
      });
    }

    if (vitals.sleepHours < 6.0 || vitals.moodScore < 6.0 || vitals.cortisol > 18) {
      actions.push({
        id: 'act-circadian',
        priority: 'COUNTERMEASURE',
        system: 'Behavioral / Endocrine',
        title: 'Activate 480nm Blue-Enriched SLA Lighting & VR Terrestrial Re-Anchoring',
        detail: `Sleep deficit (${vitals.sleepHours} hrs) and elevated salivary cortisol (${vitals.cortisol} µg/dL) confirm circadian phase drift and confinement stress.`,
        linkedTaskId: 'sla_light',
        badgeColor: 'bg-sky-500/20 text-sky-300 border-sky-500/50'
      });
    }

    if (vitals.systolic >= 130 || vitals.glucose >= 105) {
      actions.push({
        id: 'act-cardio',
        priority: 'PREVENTIVE',
        system: 'Cardio / Metabolic',
        title: 'Execute 30-Min Aerobic Interval Protocol on T2 Treadmill',
        detail: `BP (${vitals.systolic}/${vitals.diastolic} mmHg) and fasting glucose (${vitals.glucose} mg/dL) reflect microgravity insulin resistance and vascular stiffening.`,
        linkedTaskId: 't2',
        badgeColor: 'bg-emerald-500/20 text-emerald-300 border-emerald-500/50'
      });
    }

    if (actions.length === 0) {
      actions.push({
        id: 'act-nom',
        priority: 'ROUTINE MAINTENANCE',
        system: 'All Systems Nominal',
        title: 'Maintain Standard 2.5-Hour Daily Countermeasure Block',
        detail: 'All physiological indicators are tracking within personal pre-flight baseline envelopes.',
        linkedTaskId: 'ared',
        badgeColor: 'bg-emerald-500/20 text-emerald-300 border-emerald-500/50'
      });
    }

    return actions;
  }, [vitals]);

  const selectedSystem = useMemo(
    () => BODY_SYSTEMS.find(s => s.id === selectedSystemId) || BODY_SYSTEMS[0],
    [selectedSystemId]
  );

  // Metrics breakdown for the currently selected body system
  const selectedSystemMetrics = useMemo(() => {
    switch (selectedSystemId) {
      case 'neuro':
        return [
          {
            label: 'Intracranial Pressure (ICP)',
            value: `${vitals.icp} mmHg`,
            baseline: `${PERSONAL_BASELINE.icp} mmHg`,
            delta: `${(vitals.icp - PERSONAL_BASELINE.icp >= 0 ? '+' : '')}${(vitals.icp - PERSONAL_BASELINE.icp).toFixed(1)}`,

            unitNote: 'Non-invasive tympanic/ocular estimate'
          },
          {
            label: 'Optic Nerve Sheath (ONSD)',
            value: `${vitals.opticNerveDiameter} mm`,
            baseline: `${PERSONAL_BASELINE.opticNerveDiameter} mm`,
            delta: `${(vitals.opticNerveDiameter - PERSONAL_BASELINE.opticNerveDiameter >= 0 ? '+' : '')}${(vitals.opticNerveDiameter - PERSONAL_BASELINE.opticNerveDiameter).toFixed(1)}`,

            value: `${vitals.sleepHours} hrs`,

            delta: `${(vitals.sleepHours - PERSONAL_BASELINE.sleepHours >= 0 ? '+' : '')}${(vitals.sleepHours - PERSONAL_BASELINE.sleepHours).toFixed(1)}`,

            label: 'Mood & Cognitive Readiness',
            value: `${vitals.moodScore} / 10`,

                      {task.completed && <Check className="w-3.5 h-3.5 stroke-[3]" />}
                    </div>
                    <div className="flex-1 min-w-0">
                      <div className="flex items-center justify-between gap-2">
                        <span
                          className={`text-xs sm:text-sm font-semibold ${
                            task.completed ? 'line-through text-slate-400' : 'text-white'
                          }`}
                        >
                          {task.title}
                        </span>
                        <span className="text-[10px] font-mono px-2 py-0.5 rounded bg-sky-500/15 text-sky-300 shrink-0">
                          +{task.xp} XP
                        </span>
                      </div>
                      <div className="flex flex-wrap items-center justify-between text-[11px] font-mono text-slate-400 mt-1">
                        <span>{task.duration} • {task.category}</span>
                        <span className="text-emerald-400/90">{task.impactText}</span>
                      </div>
                    </div>
                  </div>
                ))}
              </div>
            </div>

            {/* Ground Flight Surgeon Async Packet Status */}
            <div className="p-3.5 rounded-xl bg-slate-950/80 border border-slate-800 flex items-center justify-between">
              <div className="flex items-center gap-2.5">
                <UserCheck className="w-5 h-5 text-sky-400" />
                <div>
                  <div className="text-xs font-semibold text-white">
                    Encrypted Store-and-Forward Medical Telemetry Packet
                  </div>
                  <div className="text-[11px] font-mono text-slate-400">
                    Auto-queued for Johnson Space Center Flight Surgeon ({SCENARIOS[activeScenario].commsDelay} delay)
                  </div>
                </div>
              </div>
              <button
                onClick={() => {
                  const now = new Date();
                  setLastSyncTime(now.toTimeString().split(' ')[0] + ' UTC');
                  setNotificationBanner('Telemetry burst packet queued for Deep Space Network (DSN) uplink.');
                  setTimeout(() => setNotificationBanner(null), 3500);
                }}
                className="px-3 py-1.5 rounded-lg bg-slate-800 hover:bg-slate-700 text-sky-300 text-xs font-mono transition cursor-pointer shrink-0"
              >
                Transmit Packet
              </button>
            </div>
          </div>
        </section>
      </main>

      {/* ================================================================= */}
      {/* MODAL 1: MANUAL VITALS & SELF-ASSESSMENT LOGGING DRAWER           */}
      {/* ================================================================= */}
      {isLogModalOpen && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#0F172A] border border-slate-700 rounded-2xl max-w-xl w-full p-6 space-y-5 shadow-2xl">
            <div className="flex items-center justify-between border-b border-slate-800 pb-3">
              <div>
                <span className="text-xs font-mono uppercase text-sky-400 block">
                  CREW SELF-MONITORING INPUT
                </span>
                <h3 className="text-lg font-bold text-white">
                  Log Biometric Vitals & Behavioral State
                </h3>
              </div>
              <button
                onClick={() => setIsLogModalOpen(false)}
                className="p-1.5 rounded-lg bg-slate-800 text-slate-400 hover:text-white cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 text-xs">
              {/* Systolic BP */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Systolic BP (mmHg)</span>
                  <span className="text-sky-400 font-bold">{vitals.systolic} mmHg</span>
                </div>
                <input
                  type="range"
                  min="95"
                  max="165"
                  value={vitals.systolic}
                  onChange={e => setVitals({ ...vitals, systolic: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Heart Rate */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Resting Heart Rate (bpm)</span>
                  <span className="text-sky-400 font-bold">{vitals.heartRate} bpm</span>
                </div>
                <input
                  type="range"
                  min="45"
                  max="110"
                  value={vitals.heartRate}
                  onChange={e => setVitals({ ...vitals, heartRate: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Intracranial Pressure */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Intracranial Pressure (ICP)</span>
                  <span className="text-sky-400 font-bold">{vitals.icp} mmHg</span>
                </div>
                <input
                  type="range"
                  min="9"
                  max="24"
                  step="0.2"
                  value={vitals.icp}
                  onChange={e => setVitals({ ...vitals, icp: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Fasting Blood Glucose */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Blood Sugar (mg/dL)</span>
                  <span className="text-sky-400 font-bold">{vitals.glucose} mg/dL</span>
                </div>
                <input
                  type="range"
                  min="70"
                  max="150"
                  value={vitals.glucose}
                  onChange={e => setVitals({ ...vitals, glucose: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Daily Sleep Hours */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Sleep Duration (hrs)</span>
                  <span className="text-sky-400 font-bold">{vitals.sleepHours} hrs</span>
                </div>
                <input
                  type="range"
                  min="3.0"
                  max="9.5"
                  step="0.2"
                  value={vitals.sleepHours}
                  onChange={e => setVitals({ ...vitals, sleepHours: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Mood / Cognitive Readiness */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Mood & Cognitive State (1-10)</span>
                  <span className="text-sky-400 font-bold">{vitals.moodScore} / 10</span>
                </div>
                <input
                  type="range"
                  min="1"
                  max="10"
                  step="0.2"
                  value={vitals.moodScore}
                  onChange={e => setVitals({ ...vitals, moodScore: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Urinary Calcium */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Urinary Calcium (mg/day)</span>
                  <span className="text-sky-400 font-bold">{vitals.urinaryCalcium} mg/d</span>
                </div>
                <input
                  type="range"
                  min="120"
                  max="360"
                  step="5"
                  value={vitals.urinaryCalcium}
                  onChange={e => setVitals({ ...vitals, urinaryCalcium: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>

              {/* Radiation Rate */}
              <div className="p-3 rounded-xl bg-slate-900 border border-slate-800">
                <div className="flex justify-between font-mono mb-1">
                  <span className="text-slate-300">Radiation Flux (mSv/day)</span>
                  <span className="text-sky-400 font-bold">{vitals.dailyRadRate} mSv/d</span>
                </div>
                <input
                  type="range"
                  min="0.2"
                  max="6.0"
                  step="0.1"
                  value={vitals.dailyRadRate}
                  onChange={e => setVitals({ ...vitals, dailyRadRate: Number(e.target.value) })}
                  className="w-full accent-sky-400 cursor-pointer"
                />
              </div>
            </div>

            <div className="flex items-center justify-end gap-3 pt-3 border-t border-slate-800">
              <button
                onClick={() => setVitals({ ...SCENARIOS[activeScenario].vitals })}
                className="px-4 py-2 rounded-lg bg-slate-800 text-slate-300 text-xs font-mono hover:bg-slate-700 cursor-pointer"
              >
                Reset to Scenario Default
              </button>
              <button
                onClick={() => {
                  setIsLogModalOpen(false);
                  setNotificationBanner('Self-assessment vitals logged. Autonomous countermeasure engine updated.');
                  setTimeout(() => setNotificationBanner(null), 4000);
                }}
                className="px-5 py-2 rounded-lg bg-sky-500 hover:bg-sky-400 text-slate-950 font-semibold text-xs font-mono cursor-pointer"
              >
                Save & Recalculate Risk
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ================================================================= */}
      {/* MODAL 2: INTERACTIVE GUIDED SELF-EXAM DIAGNOSTIC WORKFLOW         */}
      {/* ================================================================= */}
      {isExamModalOpen && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-[#0F172A] border border-sky-500/40 rounded-2xl max-w-lg w-full p-6 space-y-5 shadow-2xl">
            <div className="flex items-center justify-between border-b border-slate-800 pb-3">
              <div>
                <span className="text-xs font-mono uppercase text-sky-400 block">
                  AUTONOMOUS CREW MEDICAL DIAGNOSTIC
                </span>
                <h3 className="text-base font-bold text-white">{selectedSystem.selfExamTitle}</h3>
              </div>
              <button
                onClick={() => setIsExamModalOpen(false)}
                className="p-1.5 rounded-lg bg-slate-800 text-slate-400 hover:text-white cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="space-y-3">
              {selectedSystem.selfExamSteps.map((step, idx) => {
                const isDone = idx <= examStepProgress;
                return (
                  <div
                    key={idx}
                    onClick={() => setExamStepProgress(idx)}
                    className={`p-3.5 rounded-xl border transition cursor-pointer flex items-start gap-3 ${
                      isDone
                        ? 'bg-sky-500/15 border-sky-400/50 text-white'
                        : 'bg-slate-900 border-slate-800 text-slate-400'
                    }`}
                  >
                    <div
                      className={`w-6 h-6 rounded-full font-mono text-xs flex items-center justify-center shrink-0 ${
                        isDone ? 'bg-sky-400 text-slate-950 font-bold' : 'bg-slate-800 text-slate-400'
                      }`}
                    >
                      {idx + 1}
                    </div>
                    <div className="text-xs sm:text-sm leading-relaxed">{step}</div>
                  </div>
                );
              })}
            </div>

            <div className="flex items-center justify-between pt-3 border-t border-slate-800">
              <span className="text-xs font-mono text-slate-400">
                Step {examStepProgress + 1} of {selectedSystem.selfExamSteps.length}
              </span>
              <div className="flex gap-2">
                {examStepProgress < selectedSystem.selfExamSteps.length - 1 ? (
                  <button
                    onClick={() => setExamStepProgress(p => p + 1)}
                    className="px-4 py-2 rounded-lg bg-sky-500 hover:bg-sky-400 text-slate-950 font-semibold text-xs font-mono cursor-pointer"
                  >
                    Next Calibration Step
                  </button>
                ) : (
                  <button
                    onClick={() => {
                      setIsExamModalOpen(false);
                      setNotificationBanner(`${selectedSystem.shortName} self-exam telemetry verified and calibrated.`);
                      setTimeout(() => setNotificationBanner(null), 4000);
                    }}
                    className="px-4 py-2 rounded-lg bg-emerald-500 hover:bg-emerald-400 text-slate-950 font-semibold text-xs font-mono cursor-pointer"
                  >
                    Complete & Sync Sensor Data
                  </button>
                )}
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
